1. **`_populate_task_reminder_list()`** (line 1605):
   ```python
   def _populate_task_reminder_list(self):
       self._reminder_list.clear()
       parent = self.parent()
       if parent is None or not hasattr(parent, '_timer_manager'):
           return
       now_ts = datetime.now().timestamp()
       for cfg in parent._timer_manager.to_list():
           if cfg.get("taskId") is None:
               continue  # Skip generic timers
           if not cfg.get("enabled", True):
               continue
           trigger_at = cfg.get("nextTriggerAt", 0)
           if trigger_at <= now_ts:
               continue  # Already fired
           remaining = int(trigger_at - now_ts)
           hours, remainder = divmod(remaining, 3600)
           minutes, secs = divmod(remainder, 60)
           if hours > 0:
               time_str = f"{hours}h {minutes}m"
           elif minutes > 0:
               time_str = f"{minutes}m {secs}s"
           else:
               time_str = f"{secs}s"
           name = cfg.get("name", "Task reminder")
           label = f"{name[:60]} — {time_str}"
           item = QListWidgetItem(label)
           item.setData(Qt.ItemDataRole.UserRole, cfg.get("taskId"))
           self._reminder_list.addItem(item)
   ```

2. **`_clear_task_reminder_from_list(item)`** (line 1630):

   def _clear_task_reminder_from_list(self, item):
       task_id = item.data(Qt.ItemDataRole.UserRole)
       parent = self.parent()
       if parent is not None and hasattr(parent, '_timer_manager'):
           parent._timer_manager.cancel_task_reminder(task_id)
           parent.app_state["timers"] = parent._timer_manager.to_list()
           parent.state_manager.save()
           self._populate_task_reminder_list()  # Refresh
   ```

3. **`_edit_task_reminder_from_list(item)`** (line 1641):

   def _edit_task_reminder_from_list(self, item):
       task_id = item.data(Qt.ItemDataRole.UserRole)
       parent = self.parent()
       if parent is None:
           return
       task = next((t for t in parent.tasks if t["id"] == task_id), None)
       if task is not None:
           parent._show_custom_reminder_dialog(task)
           self._populate_task_reminder_list()  # Refresh after edit
   ```
