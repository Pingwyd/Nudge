# S5-F1-C1: Create DrawerWidget Class

**Sprint:** 5 — Reminders Drawer
**Type:** Structural
**Model:** MiMo

---

**OBJECTIVE:** Create a `DrawerWidget` class that slides in from the left side of the main window with smooth animation.

**FILE:** `src/frontend/main_window.py`

**IMPLEMENTATION:**

```python
class DrawerWidget(QWidget):
    """Sliding drawer that shows pending reminders."""
    
    def __init__(self, parent=None):
        super().__init__(parent)
        self._main_window = parent
        self._is_open = False
        
        # Width constraints
        self.setMinimumWidth(0)
        self.setMaximumWidth(0)
        self.setFixedHeight(parent.height() if parent else 400)
        
        # Animation
        self._animation = QPropertyAnimation(self, b"maximumWidth")
        self._animation.setDuration(250)
        self._animation.setEasingCurve(QEasingCurve.Type.InOutCubic)
        
        # Content
        self._init_ui()
    
    def _init_ui(self):
        layout = QVBoxLayout(self)
        layout.setContentsMargins(12, 12, 12, 12)
        
        # Header
        header = QHBoxLayout()
        title = QLabel("Reminders")
        title.setStyleSheet("font-size: 16px; font-weight: bold;")
        close_btn = QPushButton("✕")
        close_btn.setFixedSize(24, 24)
        close_btn.clicked.connect(self.close_drawer)
        header.addWidget(title)
        header.addStretch()
        header.addWidget(close_btn)
        layout.addLayout(header)
        
        # Scroll area for reminder list
        self._scroll = QScrollArea()
        self._scroll.setWidgetResizable(True)
        self._scroll.setHorizontalScrollBarPolicy(Qt.ScrollBarPolicy.ScrollBarAlwaysOff)
        self._scroll_content = QWidget()
        self._list_layout = QVBoxLayout(self._scroll_content)
        self._list_layout.setAlignment(Qt.AlignmentFlag.AlignTop)
        self._scroll.setWidget(self._scroll_content)
        layout.addWidget(self._scroll)
    
    def open_drawer(self):
        if self._is_open:
            return
        self._is_open = True
        self.show()
        self._animation.setStartValue(0)
        self._animation.setEndValue(300)
        self._animation.start()
    
    def close_drawer(self):
        if not self._is_open:
            return
        self._animation.setStartValue(self.maximumWidth())
        self._animation.setEndValue(0)
        self._animation.finished.connect(self._on_close_finished)
        self._animation.start()
    
    def _on_close_finished(self):
        self._is_open = False
        self.hide()
        self._animation.finished.disconnect(self._on_close_finished)
    
    def toggle_drawer(self):
        if self._is_open:
            self.close_drawer()
        else:
            self.open_drawer()
```

**ACCEPTANCE CRITERIA:**
- DrawerWidget class exists in main_window.py
- Slide animation works (250ms, InOutCubic easing)
- Open/close/toggle methods functional
- Drawer starts hidden (width=0)
- No import errors or syntax issues
