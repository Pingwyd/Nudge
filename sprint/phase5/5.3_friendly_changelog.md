### Step 1 — Create `src/frontend/changelog_dialog.py`

```python
"""Dedicated changelog viewer dialog."""

from __future__ import annotations

from PyQt6.QtCore import Qt
from PyQt6.QtWidgets import QDialog, QHBoxLayout, QLabel, QPushButton, QTextBrowser, QVBoxLayout

from src import __app_name__

CHANGELOG = """
<h2>Nudge Changelog</h2>

<h3>v1.2.0 (2025-06-01)</h3>
<ul>
  <li>Auto-update with background check and download progress</li>
  <li>Configurable keyboard shortcut toggle (Ctrl+Shift+N)</li>
  <li>Export tasks to CSV/JSON</li>
  <li>Performance optimization and faster task rendering</li>
  <li>Better memory management</li>
</ul>

<h3>v1.1.0 (2025-05-15)</h3>
<ul>
  <li>System tray icon with close-to-tray</li>
  <li>Close confirmation dialog</li>
  <li>Drag-and-drop task reordering</li>
  <li>Move to Top / Move to Bottom in task context menu</li>
  <li>Improved context menu styling</li>
</ul>

<h3>v1.0.0 (2025-05-01)</h3>
<ul>
  <li>Initial release</li>
  <li>Task management with add/edit/delete</li>
  <li>Collapsible task groups</li>
  <li>Dark and light themes</li>
  <li>Resizable window with liquid-glass design</li>
  <li>History tracking with restore</li>
  <li>Always on Top and Pin to Desktop modes</li>
</ul>
"""


class ChangelogDialog(QDialog):
    """Non-modal dialog showing project changelog."""

    def __init__(self, parent=None):
        super().__init__(parent)
        self.setWindowTitle(f"{__app_name__} — What's New")
        self.setFixedSize(520, 450)
        self._setup_ui()
        self._apply_theme()

    def _setup_ui(self):
        layout = QVBoxLayout(self)
        layout.setSpacing(12)

        browser = QTextBrowser()
        browser.setHtml(CHANGELOG)
        browser.setOpenExternalLinks(True)
        layout.addWidget(browser, 1)

        btn_layout = QHBoxLayout()
        btn_layout.addStretch()
        close_btn = QPushButton("Close")
        close_btn.clicked.connect(self.accept)
        close_btn.setFixedWidth(100)
        btn_layout.addWidget(close_btn)
        layout.addLayout(btn_layout)

    def _apply_theme(self):
        from src.backend.theme import normalize_theme_id, get_theme
        theme_id = "dark"
        parent = self.parent()
        if parent and hasattr(parent, "app_state"):
            theme_id = normalize_theme_id(parent.app_state.get("theme", "dark"))
        theme = get_theme(theme_id)
        self.setStyleSheet(f"""
            QDialog {{ background: {theme["windowBg"]}; color: {theme["fg"]}; }}
            QTextBrowser {{ background: {theme["cardBg"]}; color: {theme["fg"]};
                            border: 1px solid {theme.get("border", "#333")};
                            padding: 12px; }}
            QPushButton {{ background: {theme["accent"]}; color: {theme["buttonFg"]};
                           border: none; padding: 8px 20px; border-radius: 4px; }}
        """)
```
