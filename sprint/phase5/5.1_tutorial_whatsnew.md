# Phase 5.1 — In-App Tutorial / What's New Dialog

## Objective
Create a tutorial dialog that shows on first launch or update, explaining basic usage.

## Implementation

### Step 1 — Create `src/frontend/tutorial_dialog.py`

```python
"""First-launch tutorial and What's New dialog."""

from __future__ import annotations

from PyQt6.QtCore import Qt
from PyQt6.QtWidgets import (
    QDialog, QHBoxLayout, QLabel, QPushButton, QTextBrowser, QVBoxLayout,
)

from src import __app_name__, __version__


class TutorialDialog(QDialog):
    """Shows welcome screen on first launch or What's New on version change."""

    WELCOME_TEXT = """
<h2>Welcome to Nudge!</h2>
<p>Nudge is a liquid-glass desktop task widget that stays on your screen.</p>
<ul>
  <li><b>Add tasks</b> — Type in the input box and press Enter</li>
  <li><b>Edit tasks</b> — Click "Edit" on any task row</li>
  <li><b>Reorder</b> — Drag tasks up/down, or use right-click menu</li>
  <li><b>Groups</b> — Organize tasks into collapsible groups</li>
  <li><b>History</b> — View completed/deleted tasks</li>
  <li><b>Always on Top</b> — Pin Nudge above all windows</li>
</ul>
<p>Right-click any task for more options.</p>
    """

    WHATS_NEW_TEMPLATE = """
<h2>What's New in v{version}</h2>
<p>Here's what changed:</p>
<ul>
  <li>{changes}</li>
</ul>
    """

    def __init__(self, mode="welcome", version="", changes="", parent=None):
        super().__init__(parent)
        self.setWindowTitle(f"{__app_name__} — {mode.title()}")
        self.setFixedSize(480, 400)
        self._setup_ui(mode, version, changes)
        self._apply_theme()

    def _setup_ui(self, mode, version, changes):
        layout = QVBoxLayout(self)
        layout.setSpacing(12)

        if mode == "welcome":
            content = self.WELCOME_TEXT
        else:
            content = self.WHATS_NEW_TEMPLATE.format(version=version, changes=changes)

        browser = QTextBrowser()
        browser.setHtml(content)
        browser.setOpenExternalLinks(True)
        layout.addWidget(browser, 1)

        btn_layout = QHBoxLayout()
        btn_layout.addStretch()

        close_btn = QPushButton("Got it!")
        close_btn.clicked.connect(self.accept)
        close_btn.setFixedWidth(120)
        btn_layout.addWidget(close_btn)

        layout.addLayout(btn_layout)

    def _apply_theme(self):
        from src.backend.theme import normalize_theme_id, get_theme
        theme_id = "dark"
        parent = self.parent()
        if parent and hasattr(parent, "app_state"):
            theme_id = normalize_theme_id(parent.app_state.get("theme", "dark"))
        theme = get_theme(theme_id)
        self.setStyleSheet(f"""
            QDialog {{ background: {theme["windowBg"]}; color: {theme["fg"]}; }}
            QTextBrowser {{ background: {theme["cardBg"]}; color: {theme["fg"]};
                            border: 1px solid {theme.get("border", "#333")};
                            padding: 12px; }}
            QPushButton {{ background: {theme["accent"]}; color: {theme["buttonFg"]};
                           border: none; padding: 8px 20px; border-radius: 4px; }}
        """)
```

### Step 2 — Wire into main_window.py `__init__`

```python
def _check_first_launch(self):
    """Show welcome on first launch, What's New on version change."""
    last_version = self.app_state.get("lastLaunchedVersion", "")
    if not last_version:
        from src.frontend.tutorial_dialog import TutorialDialog
        dlg = TutorialDialog(mode="welcome", parent=self)
        dlg.exec()
    elif last_version != __version__ and hasattr(self, '_get_changelog_since'):
        changes = self._get_changelog_since(last_version)
        if changes:
            from src.frontend.tutorial_dialog import TutorialDialog
            dlg = TutorialDialog(mode="whatsnew", version=__version__,
                                  changes=changes, parent=self)
            dlg.exec()
    self.app_state["lastLaunchedVersion"] = __version__
```

Call `self._check_first_launch()` at end of `__init__`.

### Step 3 — Add changelog lookup helper

```python
def _get_changelog_since(self, old_version):
    """Return human-readable changes between versions."""
    # Simple version-based lookup; could read from a CHANGELOG file
    changelog = {
        "1.0.0": "Initial release with task management, groups, dark/light themes.",
        "1.1.0": "System tray, close confirmation, drag reorder, context menu improvements.",
        "1.2.0": "Auto-update, configurable shortcuts, export dialog, performance optimization.",
    }
    entries = []
    for ver, note in changelog.items():
        if self._version_gt(ver, old_version):
            entries.append(f"<b>v{ver}</b>: {note}")
    return "<br>".join(entries) if entries else ""
```
